#!/bin/bash
. ~/.bashrc
module load python/3.11.6

# Update the variable RUN for the run that you are processing. 
# Update WRKDIR variable as well.

RUN=XXXX_0000_XX
export RUN

# Dynamically extract Pawsey account from this config file location.
# This is safer than using pwd because the backup scripts source this file.
CONFIG_DIR=$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd -P)
export PAWSEY_ACCOUNT=$(echo "$CONFIG_DIR" | grep -o '/scratch/pawsey[0-9]\+' | cut -d/ -f3)
if [[ -z "$PAWSEY_ACCOUNT" ]]; then
  echo "ERROR: Could not determine PAWSEY_ACCOUNT from config path: $CONFIG_DIR" >&2
  return 1 2>/dev/null || exit 1
fi

export WRKDIR=/scratch/$PAWSEY_ACCOUNT/$USER/$RUN

# Dynamically extract date from RUN
DATE=$(echo "$RUN" | cut -d'_' -f2)
export DATE

# Update the location of where you would like your output, or leave as is.
export rundir=$WRKDIR/draftgenomes

# Update to the location of your Git clone, or make sure it is in your WRKDIR
export scripts=$WRKDIR/OceanOmics-draft-genomes


# Leave these variables as is as the scripts use these locations based on the above variables.

export results=./
export download=$WRKDIR/basespace/$RUN
export pooled=$WRKDIR/pooled/$RUN
